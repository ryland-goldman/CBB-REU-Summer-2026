# Cornell Linac beam simulation — Python dependencies
#
# PREFERRED: install everything from conda-forge (this is how the reference CBB
# env was built). Note the conda package names that differ from the pip names:
#   pip `pywarpx`  -> conda `warpx`   (bundles the pywarpx Python module + executables;
#                                      there is NO conda package literally named "pywarpx")
#   ImpactTexe     -> conda `impact-t` (the serial Impact-T binary linac_rest runs
#                                      through lume-impact; not pip-installable)
# One-shot env creation (Linux cluster or macOS):
#   conda create -n CBB -c conda-forge warpx impact-t openpmd-api openpmd-viewer \
#       lume-impact openpmd-beamphysics easygdf numpy scipy matplotlib tqdm
#
# FALLBACK: `pip install -r requirements.txt` into an env that already has
# warpx/impact-t/openpmd-api from conda (those three are best from conda-forge;
# openpmd-api needs its C++ core, and ImpactTexe is a Fortran binary).
# pipeline/beam_gui.py additionally needs tkinter (ships with conda-forge python).

# Core PIC engine
pywarpx==26.4

# LUME wrapper that the four WarpX stages (cathode/gun/injector/linac_sec1) drive via a
# static YAML config + runtime overrides. Needs pywarpx (above) at import. 1.0.1 carries the
# AppliedFromFile field, FromInitialParticles distribution, install_callback, and set/get/update
# override API the stages rely on. (Develop against a local editable clone: pip install -e .)
lume-warpx==1.0.1

# openPMD I/O and visualization
openpmd-api==0.17.0
openpmd-viewer==1.11.0

# Impact-T driver + beam I/O for the linac_rest stage (sections 2–8)
lume-impact==0.11.0
openpmd-beamphysics==0.14.1

# GPT/GDF field-map reader
easygdf==2.1.1

# Scientific stack
numpy==2.4.6
scipy==1.17.1
matplotlib==3.10.9
tqdm==4.67.3
